## 개요

수많은 프로젝트를 진행하면서 버튼/인풋/셀렉트 등등 여러 컴포넌트를 제작하고 사용했지만 타이포그래피는 선언만 해놓고 정작 제대로 사용이 이루어지지 않아서 텍스트 구조를 일관성 있게 유지하지 못했다.

그러던 도중 토스의 디자인 시스템에서 `Paragraph`라는 컴포넌트를 발견하고 해당 컴포넌트를 보면서 이제는 텍스트 구조 또한 시스템화하여 재활용하는게 좋겠다는 생각이 들었다.

## 컴포넌트 코드

텍스트 컴포넌트라서 그런지 구조는 생각보다 간단하게 이루어졌다
근데 `TDS`를 사용하면서 느낀 점은 Paragraph.Text / Paragraph.Icon 같은 형식의 컴포넌트를 자주 사용하는것을 보았다

나는 Angular를 주력으로 사용해왔기 때문에 해당 패턴은 생각치 못했는데 이번 기회에 자주 활용해볼 계획이다

```
import { ParagraphIcon } from "./icon/paragraph-icon";

import {

  ParagraphColor,

  ParagraphDisplay,

  ParagraphFontWeight,

  ParagraphTextAlign,

  ParagraphTypography,

} from "./paragraph.styles";

import { ParagraphProps } from "./paragraph.types";

import { ParagraphText } from "./text/paragraph-text";

  

export type { ParagraphIconProps } from "./icon/paragraph-icon.types";

  

function ParagraphRoot(

  props: ParagraphProps & React.HTMLAttributes<HTMLParagraphElement>,

) {

  const {

    ellipsisAfterLines,

    typography,

    fontWeight,

    color,

    display,

    textAlign,

    ...rest

  } = props;

  

  return (

    <p

      {...rest}

      style={{

        ...(typography ? ParagraphTypography[typography] : {}),

        ...(fontWeight ? ParagraphFontWeight[fontWeight] : {}),

        ...(color ? ParagraphColor[color] : {}),

        ...(display ? ParagraphDisplay[display] : {}),

        ...(textAlign ? ParagraphTextAlign[textAlign] : {}),

        ...(ellipsisAfterLines && ellipsisAfterLines > 0

          ? {

              display: "-webkit-box",

              WebkitBoxOrient: "vertical",

              WebkitLineClamp: Math.floor(ellipsisAfterLines),

              overflow: "hidden",

            }

          : {}),

      }}

    >

      {props.children}

    </p>

  );

}

  

export const Paragraph = Object.assign(ParagraphRoot, {

  Text: ParagraphText,

  Icon: ParagraphIcon,

});
```


## 사용

```
<div className="flex flex-col gap-1">

        <Paragraph typography="t2" fontWeight="bold">

          {title || ""}

          {description || ""}

        </Paragraph>

      </div>
```
